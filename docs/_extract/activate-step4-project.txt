function validate() {
 var ok = true;
 ok = setErr($('pname'), $('pname').value.trim() ? '' : '请填写项目名称') && ok;
 ok = setErr($('ptype'), sel.length ? '' : '请选择项目类型') && ok;
 ok = setErr($('acc'), $('acc').value.trim() ? '' : '请填写管理员账号') && ok;
 var p = $('pwd').value;
 var pm = !p ? '请填写管理员密码' : (p.length 16 || !/[A-Z]/.test(p) || !/[a-z]/.test(p)) ? '密码 8–16 位，必须包含大写字母和小写字母' : '';
 ok = setErr($('pwd'), pm) && ok;
 ok = setErr($('map'), point ? '' : '请在地图上选择项目中心') && ok;
 return ok;
}
