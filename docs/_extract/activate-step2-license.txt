var pop = document.getElementById('pop'), curDot = null;
function openNote(dot) {
 if (curDot === dot) return closeNote();
 closeNote();
 var n = NOTES[dot.dataset.note];
 document.getElementById('popTitle').textContent = n.title;
 document.getElementById('popBody').innerHTML = n.items.map(function (it) {
 return '' + it.tag + '' + it.ver + '' + it.html + '
';
 }).join('');
 pop.classList.add('show');
 var r = dot.getBoundingClientRect();
 var left = r.right + 10 + window.scrollX, top = r.top - 10 + window.scrollY;
 if (r.right + 400 > document.documentElement.clientWidth) left = r.left - 390 + window.scrollX;
 pop.style.left = left + 'px'; pop.style.top = top + 'px';
 dot.classList.add('open'); curDot = dot;
}
function closeNote() { pop.classList.remove('show'); if (curDot) curDot.classList.remove('open'); curDot = null; }
document.addEventListener('click', function (e) {
 var d = e.target.closest('.dot');
 if (d) { e.stopPropagation(); openNote(d); return; }
 if (!pop.contains(e.target)) closeNote();
});

// ===== 上传授权文件 → 自动验证 =====
var drop = document.getElementById('drop'), picker = document.getElementById('filePicker');
var dropTxt = document.getElementById('dropTxt'), licErr = document.getElementById('licErr'), busy = false;
function verify(file) {
 if (!file || busy) return;
 busy = true; licErr.textContent = ''; drop.classList.remove('bad'); drop.classList.add('busy');
 dropTxt.textContent = '正在验证 ' + file.name + ' …';
 setTimeout(function () {
 busy = false; drop.classList.remove('busy'); dropTxt.textContent = '选择文件';
 if (!chkOk) { drop.classList.add('bad'); licErr.textContent = '该文件不正确，请重新选择授权文件'; return toast('该文件不正确'); }
 localStorage.setItem(STORE, JSON.stringify(LIC));
 toast('授权文件验证通过'); showInfo(LIC); markSt('info');
 }, 1200);
}
drop.onclick = function () { if (busy) return; picker.value = ''; picker.click(); };
picker.onchange = function () { verify(picker.files[0]); };
drop.addEventListener('dragover', function (e) { e.preventDefault(); });
drop.addEventListener('drop', function (e) { e.preventDefault(); verify(e.dataTransfer.files[0]); });
